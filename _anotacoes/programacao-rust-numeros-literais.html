Caso queiramos escrever um número temos 5 formas:

  • Decimal
    • 1_000_000
  • Hexadecimal
    • 0xff
  • Octal
    • 0o22
  • Binário
    • 0b1111_1111
  • Byte
    • b’F’
fn main() {  
  
	let decimal = 2_5_5;  
  
	let octal = 0o377;  
  
	let hexadecimal = 0xFF;  
  
    let binario = 0b1111_1111;  
  
    let byte = b'\xFF';  
  
    println!("{}\n\{}\n{}\n{}\n{}",decimal,octal,hexadecimal,binario,byte);  
  
}
